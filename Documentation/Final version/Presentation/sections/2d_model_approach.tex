\centering
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[
	node distance=1.2em and 1.6em,
	block/.style={draw, rounded corners, align=center, inner sep=4pt},
	arrow/.style={-{Stealth[length=2.2mm]}, thick}
]

\node[block] (input) {History window $u_{t-W+1:t}$\\$B\times W\times C\times H\times W$};
\node[block, right=of input] (attn) {Causal Temporal\\Attention (2D)};
\node[block, right=of attn] (film) {FiLM conditioning\\from $\nu$};
\node[block, right=of film] (decoder) {Conv2D decoder};
\node[block, right=of decoder] (tanh) {Tanh clip\\$\Delta$};
\node[block, right=of tanh] (output) {Next field\\$u_{t+1}$};

\draw[arrow] (input) -- (attn);
\draw[arrow] (attn) -- (film);
\draw[arrow] (film) -- (decoder);
\draw[arrow] (decoder) -- (tanh);
\draw[arrow] (tanh) -- (output);

\node[block, below=0.9em of film] (nu) {Viscosity $\nu$\\(scalar)};
\draw[arrow] (nu) -- (film);

\end{tikzpicture}%
