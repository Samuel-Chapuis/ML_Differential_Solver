\section{TCAN Model}

\begin{frame}{\Large TCAN: Temporal Convolutional Attention Network}

\begin{columns}[T]
\column{0.85\textwidth}
\textbf{ \small Model Definition}
\begin{itemize}
    \item \scriptsize TCAN: Temporal Convolutional Attention Network.
    \item \scriptsize Feed-forward autoregressive model (each prediction becomes input for the next step).
    \item \scriptsize Controller in NFTM framework.
\end{itemize}

\textbf{\small Continuous Field}
\begin{itemize}
    \item \scriptsize $f_t$: snapshot of PDE solution.
    \item \scriptsize Vector of size $N$ (spatial positions).
    \item \scriptsize $f_t = [u(x_1,t), \ldots, u(x_N,t)]$.
    \item \scriptsize Shape: $(1 \times N)$.
\end{itemize}

\textbf{\small Input and Output of the Model}\\
\vspace{0.1cm}
\scriptsize To predict $f_{t+1}$, the corresponding input and output of the model are:
\begin{itemize}
    \item \scriptsize \textbf{Input}: \textbf{history window} (chunk of $W$ previous fields): $[f_{t - W + 1}, f_{t - W + 2}, ..., f_{t}]$ with size $(B, W, N)$.
    \item \scriptsize \textbf{Output}: \textbf{prediction of the field at next time step $t+1$}: $\hat{f}_{t+1}$, with size: $(B, 1, N)$.
\end{itemize}

\begin{frame}{\Large TCAN: Temporal Convolutional Attention Network}
  \vspace{-0.3em}
  \begin{block}{Why TCAN works better?}
    It combines a \textbf{convolution that looks at recent
    steps closely} + \textbf{attention mechanism} that also considers
    the \textbf{full history} $\Rightarrow$ Best of CNN and Transformer worlds.
  \end{block}
\begin{columns}[t]
  \begin{column}{0.52\textwidth}
    \small \textbf{Autoregressive Roll-out Process}
    \vspace{0.1em}
    \begin{itemize}
      \item Each TCAN call predicts the next step from the current window.
      \item The window slides: drop the oldest, add the new prediction.
      \item Repeat for the full trajectory.
    \end{itemize}

    \scriptsize In the example, the model first predicts $\hat{f}_{21}$
    from $[f_1,\dots,f_{20}]$, then uses $[f_2,\dots,f_{20},\hat{f}_{21}]$
    to predict $\hat{f}_{22}$, and so on.
  \end{column}

\begin{frame}{\Large TCAN Architecture: Step by Step}
  \vspace{-0.4em}
\begin{columns}[T]
    \column{0.3\textwidth}
    \begin{center}
    \resizebox{1.25\textwidth}{!}{%
    \begin{tikzpicture}[
        box/.style={rectangle, draw, rounded corners, minimum width=2.4cm, minimum height=0.85cm, align=center, fill=blue!10},
        op/.style={rectangle, draw, rounded corners, minimum width=2.4cm, minimum height=0.85cm, align=center, fill=green!10},
        arrow/.style={->, thick, >=stealth},
        input/.style={rectangle, draw, rounded corners, minimum width=2.4cm, minimum height=1.0cm, align=center, fill=orange!20},
        output/.style={rectangle, draw, rounded corners, minimum width=2.4cm, minimum height=1.0cm, align=center, fill=red!20},
        node distance=2.0cm and 1.6cm
    ]
    % Main flow - left column
    \node[input] (input) {INPUT\\$f_{\text{history}}$};
    \node[op, below=of input] (embed) {Embedding\\Conv1d+GELU};
    \node[box, below=of embed] (features) {Features\\$(B,W,32,N)$};

    % Attention column - center-right
    \node[box, right=1.8cm of features] (query) {Query $Q$\\(last frame)};
    \node[op, below=of query] (attn) {Causal\\Attention};
    \node[box, below=of attn] (context) {Context\\$(B,32,N)$};

    % Viscosity conditioning node
    \node[op, below=of context, fill=purple!15] (cond) {$\nu$ conditioning\\$\gamma \cdot h + \beta$};

    % Viscosity input node
    \node[input, right=1.8cm of cond, fill=teal!20] (nu) {INPUT\\$\nu$};

    % Decoder/output column
    \node[op, right=1.8cm of attn] (decoder) {Decoder\\Conv1d};
    \node[box, below=of decoder] (corr) {Correction\\$\tanh(\times0.1)$};
    \node[output, below=of corr] (output) {OUTPUT\\$f_{\text{next}}$};

    % ARROWS
    \draw[arrow] (input) -- (embed);
    \draw[arrow] (embed) -- (features);
    \draw[arrow] (features) -| ([xshift=-0.3cm]query.west) -- (query);
    \draw[arrow] (query) -- (attn);
    \draw[arrow] (attn) -- (context);
    \draw[arrow] (context) -- (cond);
    \draw[arrow] (cond) -| ([xshift=-0.3cm]decoder.west) -- (decoder);
    \draw[arrow] (decoder) -- (corr);
    \draw[arrow] (corr) -- (output);

    % Nu -> conditioning arrow
    \draw[arrow, teal!70!black, thick] (nu.west) -- (cond.east)
        node[midway, above, font=\scriptsize, black]{$\nu$};

    % Residual from features to context
    \draw[arrow, dashed, blue] (features.south)
        to[out=-5,in=150] node[above, font=\footnotesize, black]{residual}
        (context.west);

    % Residual from input to output
    \draw[arrow, dashed, orange, thick] (input.south)
        |- ++(-2,0)
        |- ++(0,-24)
        -| (output.west)
        node[below, font=\scriptsize, pos=0.1]{$f_t$};

    % KV note
    \node[font=\footnotesize, align=center, right=0.2cm of query, anchor=west] (kv) {$K,V$\\all frames};
    \end{tikzpicture}%
    }
    \end{center}

    \column{0.58\textwidth}
    \begin{itemize}
      \item \scriptsize \textbf{$f_{\text{history}}$}: the last $W$ fields the model has seen, size $(B,W,N)$.
      \item \scriptsize \textbf{Conv1d+GELU}: each field is projected into 32 internal features.
      \item \scriptsize \textbf{Query $Q$}: extracted from the most recent field only --- what we want to predict next.
      \item \scriptsize $\mathbf{K},\mathbf{V}$: extracted from all $W$ fields --- the full history the model can look at.
      \item \scriptsize \textbf{Causal Attention}: for each spatial position, the model decides how much to look at each past step: $\alpha_{w,n}=\text{softmax}_w(QK^T/\sqrt{32})$.
      \item \scriptsize \textbf{$\nu$ conditioning}: viscosity is as input: so the model behaves differently for smooth vs sharp flows.
      \item \scriptsize \textbf{Decoder}: translates the conditioned features into a small correction $\Delta f$.
      \item \scriptsize \textbf{$f_{\text{next}}$}: we add the correction to the last known field.
    \end{itemize}
  \end{columns}
\end{frame}
