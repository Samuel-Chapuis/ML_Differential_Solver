\section{1D Burgers Equation Dataset}

%% ---------------------------------------------------------------
%% SLIDE 1: Dataset structure
%% ---------------------------------------------------------------
\begin{frame}{\Large Dataset: 1D Burgers Equation Trajectories}
\vspace{-0.5em}
  \begin{columns}[t]
    \begin{column}{0.55\textwidth}
      \begin{block}{\centering \small Where does the data come from?}
        We use the \textbf{official dataset} from the FNO paper
        (Li et al., 2021).\\ The trajectories were generated by
        numerically solving the 1D Burgers equation and saving
        the full solution over time.
        \begin{itemize}\setlength\itemsep{1pt}
          \item \small \textbf{Domain}: $x \in [0,1]$, $t \in [0,1]$.
          \vspace{0.2em}
          \item \textbf{Periodic boundary conditions}.
          \vspace{0.2em}
          \item \small $\nu \in [0,1]$.
          \vspace{0.2em}
          \item \textbf{1000 train + 200 test} per resolution.
          \vspace{0.05em}
          \item Four grid resolutions: $s \times s$ with $s \in \{85, 141, 211, 421\}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{\centering \small Initial conditions}
        \begin{itemize}
            \item \small $u(x,0)$ is randomly generated for each trajectory.
              \item \small \textbf{Random Fourier modes}.
              \item \small \textbf{Single sine waves}.
              \item \small \textbf{Two-mode sines}.
            \item This variety ensures the model learns to handle a wide range of initial conditions, from simple to complex.
        \end{itemize}
        \end{block}
    \end{column}
  \end{columns}

\end{frame}

%% SLIDE 3: Effect of viscosity explained
%% ---------------------------------------------------------------
\begin{frame}{\Large Effect of Viscosity $\nu$}
\vspace{-1.2em}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\centering Low viscosity ($\nu = 0.001$)}
        \begin{itemize}\setlength\itemsep{4pt}
          \item Nothing is smoothing the wave.
          \item It keeps steepening into a sharp drop.
          \item Very sensitive to the starting shape.
          \item \textbf{Hard for the model}: things change fast.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\centering High viscosity ($\nu = 0.3$)}
        \begin{itemize}\setlength\itemsep{4pt}
          \item The wave is constantly being smoothed out.
          \item It slowly shrinks and flattens over time.
          \item Predictable, no sudden changes.
          \item \textbf{Easier for the model}: slow, regular behaviour.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2em}

  \begin{alertblock}{Why does this matter?}
    The model needs to handle both cases. This is why we give it
    $\nu$ as an input, so it knows in advance what kind of
    behaviour to expect.
  \end{alertblock}

\end{frame}

%% SLIDE 4: Multi-resolution evaluation
%% ---------------------------------------------------------------
\begin{frame}{\Large Testing at 4 Different Resolutions}
\vspace{-0.6em}
  \begin{block}{Why 4 resolutions?}
    \begin{itemize}
      \item Want to check if the model works well no matter how many
    grid points we use.
    \item We trained one model per resolution, all with the same architecture, and compared the results.
    \end{itemize}
  \end{block}
\vspace{-1.7em}
  \begin{columns}[t]
    \begin{column}{0.55\textwidth}
      \begin{block}{\centering The challenge}
        \begin{itemize}\setlength\itemsep{4pt}
          \item More grid points means more time steps.
          \item More time steps means the model has to
            predict further into the future.
          \item Small errors add up: so \textbf{higher resolution is harder} for TCAN.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{block}{\centering Viscosity in the data}
        \begin{itemize}\setlength\itemsep{4pt}
          \item Some samples have shocks ($\nu \approx 0$).
          \item Some are in between.
          \item Some are fully smooth ($\nu \approx 1$).
          \item The model sees all three during training.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

\end{frame}
