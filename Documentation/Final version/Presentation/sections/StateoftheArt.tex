\section{State Of the Art}

\begin{frame}{State of the Art}
    \textcolor{red_unipd}{\large Key Papers}
    \vspace{0.4em}

    \centering
    \fontsize{6.5}{9}\selectfont
    \begin{tabularx}{\textwidth}{|p{2.6cm}|X|X|}
        \hline
        \textbf{Paper} & \textbf{Key Contribution} & \textbf{Relevance to Our Work} \\
        \hline
        \href{https://arxiv.org/abs/2010.08895}{\textbf{Fourier Neural Operator (FNO)}} \newline {\color{gray}\tiny Primary Benchmark } &
        Learns PDE solution operators via global convolution in Fourier space, up to 1000$\times$ faster than traditional solvers &
        \textbf{Our primary benchmark}  Our models are directly evaluated against this \\
        \hline
        \href{https://arxiv.org/pdf/2111.03794}{\textbf{Physics-Informed Neural Operator (PINO)}} \newline {\color{gray}\tiny Generalization } &
        Combines neural operator with physics informed constraints for parameter-to-solution generalization &
        Relevant to viscosity generalization without explicit conditioning \\
        \hline
        \href{https://arxiv.org/pdf/2505.20721}{\textbf{Recurrent Neural Operators (RNO)}} \newline {\color{gray}\tiny Rollout Stability } &
        Trains neural operators autoregressively, exposing the model to its own prediction errors to eliminate train-test mismatch &
        Directly addresses error accumulation  \\
        \hline
        \href{https://arxiv.org/pdf/2308.05732}{\textbf{PDE-Refiner}} \newline {\color{gray}\tiny Shock Recovery } &
        Iteratively refines predictions via a denoising-inspired process to recover high-frequency content lost &
        Post-processing step to recover shock sharpness degraded during autoregressive rollout \\
        \hline
    \end{tabularx}
\end{frame}

    \begin{tabularx}{\textwidth}{|p{2.6cm}|X|X|}
        \hline
        \textbf{Paper} & \textbf{Key Contribution} & \textbf{Relevance to Our Work} \\
        \hline
        \href{https://arxiv.org/pdf/2507.02227}{\textbf{PhysicsCorrect}} \newline {\color{gray}\tiny A Training-Free Approach } &
        Projects predictions onto physically consistent manifolds via PDE residual Jacobian at inference &
        Training-free correction, error reduction without any architectural changes \\
        \hline
        \href{https://arxiv.org/pdf/2402.02366}{\textbf{Transolver}} \newline {\color{gray}\tiny Spatial Attention } &
        Physics Attention groups mesh points into learned physical states rather than applying attention directly to spatial points (linear complexity) &
        Gradient structure capture beyond temporal attention \\
        \hline
        \href{https://arxiv.org/pdf/2110.09813}{\textbf{Multi-Objective Loss Balancing}} \newline {\color{gray}\tiny Training Stability } &
        Adaptive algorithms that automatically weight competing physics-informed loss terms to prevent gradient pathologies during training &
        Accelerate convergence and reduce instability \\
        \hline
        \href{https://arxiv.org/pdf/2408.01536}{\textbf{Active Learning for Neural PDE Solvers}} \newline {\color{gray}\tiny Data Efficiency } &
        Benchmark framework showing active learning via intelligent selection of PDE parameters &
        Reduces data requirements for the 2D Navier-Stokes extension\\
        \hline
    \end{tabularx}
\end{frame}
